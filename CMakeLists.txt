cmake_minimum_required(VERSION 3.24)
project(0xBOF VERSION 1.0.1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ECL_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

find_package(OpenSSL 3.2 REQUIRED COMPONENTS Crypto)
find_package(Boost 1.74 CONFIG REQUIRED)
find_package(nlohmann_json 3.10 CONFIG REQUIRED)

add_library(ecl_core src/ecl.cpp src/privacy.cpp src/crypto.cpp src/io.cpp)
target_include_directories(ecl_core PUBLIC include PRIVATE src)
target_link_libraries(ecl_core PUBLIC Boost::headers nlohmann_json::nlohmann_json PRIVATE OpenSSL::Crypto)
if(MSVC)
  target_compile_options(ecl_core PUBLIC /utf-8 PRIVATE /W4 /permissive-)
else()
  target_compile_options(ecl_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(ECL_SANITIZERS AND NOT MSVC)
  target_compile_options(ecl_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ecl_core PUBLIC -fsanitize=address,undefined)
endif()

add_executable(ecl src/main.cpp)
set_target_properties(ecl PROPERTIES OUTPUT_NAME 0xbof)
target_link_libraries(ecl PRIVATE ecl_core)
add_executable(ecl_benchmark src/benchmark.cpp)
set_target_properties(ecl_benchmark PROPERTIES OUTPUT_NAME 0xbof-benchmark)
target_link_libraries(ecl_benchmark PRIVATE ecl_core)
if(MINGW)
  target_link_options(ecl PRIVATE -municode)
  target_link_options(ecl_benchmark PRIVATE -municode)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(ecl_tests tests/test_ecl.cpp)
  target_include_directories(ecl_tests PRIVATE src)
  target_link_libraries(ecl_tests PRIVATE ecl_core)
  add_test(NAME legacy_format COMMAND ecl_tests)
  set_tests_properties(legacy_format PROPERTIES TIMEOUT 300)
endif()
install(TARGETS ecl ecl_benchmark RUNTIME DESTINATION bin)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION share/0xBOF)
